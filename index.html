/******************************************************* AP SCANNER Backend Apps Script Version 1.0 *******************************************************/ const SHEET_NAME = "SCAN_AP"; // GANTI DENGAN API KEY GOOGLE VISION const API_KEY = "MASUKKAN_API_KEY_DISINI"; /******************************************************* POST REQUEST DARI GITHUB *******************************************************/ function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (!body.image) { return jsonResponse({ success: false, message: "Image kosong" }); } const nomor = scanVision(body.image); if (!nomor) { return jsonResponse({ success: false, message: "Nomor AP tidak ditemukan" }); } const result = saveDocument(nomor); return jsonResponse({ success: true, nomor: nomor, message: result }); } catch (err) { return jsonResponse({ success: false, message: err.toString() }); } } /******************************************************* GOOGLE VISION OCR *******************************************************/ function scanVision(base64) { const url = "https://vision.googleapis.com/v1/images:annotate?key=" + API_KEY; const payload = { requests: [ { image: { content: base64 }, features: [ { type: "DOCUMENT_TEXT_DETECTION" } ] } ] }; const options = { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true }; const response = UrlFetchApp.fetch(url, options); const json = JSON.parse(response.getContentText()); Logger.log(response.getContentText()); // Jika Vision API mengembalikan error if (json.error) { throw new Error(json.error.message); } if ( !json.responses || !json.responses[0] || !json.responses[0].fullTextAnnotation ) { return null; } const text = json.responses[0] .fullTextAnnotation .text; Logger.log(text); // Cari nomor AP yang diawali 92 dan total 10 digit const match = text.match(/\b92\d{8}\b/); if (match) { return match[0]; } return null; } /******************************************************* SIMPAN KE GOOGLE SHEET *******************************************************/ function saveDocument(noAP) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { throw new Error("Sheet SCAN_AP tidak ditemukan"); } const lastRow = sheet.getLastRow(); // Cek duplikat if (lastRow > 1) { const nomorList = sheet .getRange(2, 2, lastRow - 1, 1) .getValues(); for (let i = 0; i < nomorList.length; i++) { if (String(nomorList[i][0]).trim() === String(noAP).trim()) { return "Nomor sudah pernah discan"; } } } // Simpan ke Sheet sheet.appendRow([ new Date(), noAP, "BELUM APPROVE" ]); return "Berhasil disimpan"; } /******************************************************* RESPONSE JSON *******************************************************/ function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); } /******************************************************* TEST OCR *******************************************************/ function testVision() { Logger.log("Vision API Aktif"); } /******************************************************* TEST SHEET *******************************************************/ function testSheet(){ Logger.log( saveDocument("9200009099") ); } /******************************************************* CLEAR DUPLIKAT (OPTIONAL) *******************************************************/ function clearData(){ const sheet= SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); const lastRow=sheet.getLastRow(); if(lastRow>1){ sheet .getRange( 2, 1, lastRow-1, 3 ) .clearContent(); } } /******************************************************* STATUS SERVER *******************************************************/ function doGet() { return ContentService .createTextOutput( JSON.stringify({ success: true, message: "AP Scanner API Running", version: "1.0" }) ) .setMimeType( ContentService.MimeType.JSON ); } /******************************************************* HEALTH CHECK *******************************************************/ function healthCheck(){ return { success:true, time:new Date(), sheet:SHEET_NAME, apiKey: API_KEY!="" ? "OK":"KOSONG" }; }
